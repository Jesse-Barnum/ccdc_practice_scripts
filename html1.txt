# 1. Define the Action with proper quote wrapping (' outside, " inside)
$Action = New-ScheduledTaskAction -Execute "PowerShell.exe" -Argument '-ExecutionPolicy Bypass -WindowStyle Hidden -File "C:\Program Files\Engine\ScoringEngine.ps1"'

# 2. Define the Trigger and System context
$Trigger = New-ScheduledTaskTrigger -AtStartup
$Principal = New-ScheduledTaskPrincipal -UserId "SYSTEM" -LogonType ServiceAccount -RunLevel Highest

# 3. Register the clean task
Register-ScheduledTask -TaskName "OWCAScoringEngine" -Action $Action -Trigger $Trigger -Principal $Principal -Force

# 4. Start the engine
Start-ScheduledTask -TaskName "OWCAScoringEngine"
